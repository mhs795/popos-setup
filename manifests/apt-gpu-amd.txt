# AMD Radeon RX 5700 XT (Navi 10) on the target desktop.
#
# The kernel driver (amdgpu) and firmware are already in Pop!_OS — there is no
# proprietary driver to install. These are the userspace bits that Vulkan,
# hardware video decode and 32-bit games actually need.

# Vulkan (RADV) — Steam/Proton, RPCS3, xemu, RetroArch all use it
mesa-vulkan-drivers
libvulkan1
vulkan-tools

# Hardware video decode/encode
mesa-va-drivers
mesa-vdpau-drivers
libva-drm2
vainfo

# Xorg amdgpu DDX (Wayland doesn't need it; harmless and useful as a fallback)
xserver-xorg-video-amdgpu

# 32-bit userspace — required by Steam and a lot of what Lutris runs.
# setup.sh runs "dpkg --add-architecture i386" before installing these.
mesa-vulkan-drivers:i386
libgl1-mesa-dri:i386
libvulkan1:i386

# Fan curves / clocks / undervolt GUI for Radeon cards, and sensor readout
corectrl
lm-sensors
radeontop
